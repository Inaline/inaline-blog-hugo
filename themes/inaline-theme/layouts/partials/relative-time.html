{{/* 相对时间文案（侧栏近期评论用） 入参：一个 time.Time（调用方负责解析）。 输出： · 1 分钟内 → 刚刚 · 1 小时内 → N 分钟前 · 24 小时内 → N 小时前 · 7 天内 → N 天前 · 超过 7 天 → 2006年1月2日 构建时计算一次即可：站点每次发布都会重新生成，静态页里的"X 分钟前" 与构建时刻一致。评论系统接入后若要做到页面停留时实时刷新， 再由前端接管（模板侧只留 ）。 */}} {{- $t := . -}} {{- $sec := sub now.Unix $t.Unix -}} {{- if lt $sec 0 }}{{ $sec = 0 }}{{ end -}} {{- $text := "" -}} {{- if lt $sec 60 -}} {{- $text = "刚刚" -}} {{- else if lt $sec 3600 -}} {{- $text = printf "%d 分钟前" (div $sec 60) -}} {{- else if lt $sec 86400 -}} {{- $text = printf "%d 小时前" (div $sec 3600) -}} {{- else if lt $sec 604800 -}} {{- $text = printf "%d 天前" (div $sec 86400) -}} {{- else -}} {{- $text = $t.Format "2006年1月2日" -}} {{- end -}} {{- $text -}}
